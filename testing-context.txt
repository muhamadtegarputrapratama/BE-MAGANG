<?php

namespace App\Models;

use App\Enums\StatusPengajuan;
use App\Support\Departemen;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\BelongsTo;

class PengajuanMagang extends Model
{
    protected $table = 'pengajuan_magang';

    // Nomor surat dan path file sengaja tidak di sini: diisi sistem, bukan lewat mass assignment.
    protected $fillable = [
        'nama',
        'anggota',
        'instansi',
        'alamat_instansi',
        'tanggal_mulai',
        'tanggal_selesai',
        'instansi_kedua',
        'alamat_instansi_kedua',
        'tanggal_mulai_kedua',
        'tanggal_selesai_kedua',
    ];

    protected function casts(): array
    {
        return [
            'status' => StatusPengajuan::class,
            'anggota' => 'array',
            'tanggal_mulai' => 'date',
            'tanggal_selesai' => 'date',
            'tanggal_mulai_kedua' => 'date',
            'tanggal_selesai_kedua' => 'date',
            'diverifikasi_at' => 'datetime',
            'ditandatangani_at' => 'datetime',
            'dikirim_at' => 'datetime',
        ];
    }

    public function user(): BelongsTo
    {
        return $this->belongsTo(User::class);
    }

    public function diverifikasiOleh(): BelongsTo
    {
        return $this->belongsTo(User::class, 'diverifikasi_oleh');
    }

    public function sudahDitandatangani(): bool
    {
        return $this->ditandatangani_at !== null;
    }

    /** Mahasiswa hanya boleh mengubah saat draft atau diminta revisi. */
    public function bisaDiedit(): bool
    {
        return in_array($this->status, [StatusPengajuan::Draft, StatusPengajuan::Revisi], true);
    }

    /* =========================================================
       PERINGATAN PENGAJUAN BERULANG (hanya peringatan untuk admin, tidak memblokir mahasiswa)
    ========================================================= */

    /**
     * Admin diberi peringatan bila mahasiswa (akun / NIM yang sama) sudah mengajukan sebanyak ini.
     * Bisa diubah di config/magang.php: 'ambang_peringatan_pengajuan'.
     */
    public static function ambangPeringatan(): int
    {
        return max((int) config('magang.ambang_peringatan_pengajuan', 2), 1);
    }

    /**
     * Banyaknya pengajuan yang sudah dikirim oleh seorang mahasiswa (draft tidak dihitung).
     * Revisi tetap dihitung satu: memperbaiki pengajuan tidak membuat pengajuan baru.
     */
    public static function jumlahTerkirimUser(int $userId): int
    {
        return static::where('user_id', $userId)
            ->where('status', '!=', StatusPengajuan::Draft->value)
            ->count();
    }

    /** Banyaknya pengajuan terkirim milik mahasiswa pengaju ini (termasuk pengajuan ini). */
    public function jumlahPengajuanMahasiswa(): int
    {
        return static::jumlahTerkirimUser((int) $this->user_id);
    }

    /* =========================================================
       INSTANSI (maksimal 2) -> satu instansi = satu surat
    ========================================================= */

    /** 1 bila hanya instansi pertama, 2 bila instansi kedua juga diisi. */
    public function jumlahInstansi(): int
    {
        return filled($this->instansi_kedua) ? 2 : 1;
    }

    /** Instansi tujuan berurutan; 'urutan' 1 atau 2 menentukan surat yang mana. */
    public function daftarInstansi(): array
    {
        $daftar = [[
            'urutan'  => 1,
            'nama'    => $this->instansi,
            'alamat'  => $this->alamat_instansi,
            'mulai'   => $this->tanggal_mulai,
            'selesai' => $this->tanggal_selesai,
            'nomor'   => $this->nomor_surat,
        ]];

        if ($this->jumlahInstansi() === 2) {
            $daftar[] = [
                'urutan'  => 2,
                'nama'    => $this->instansi_kedua,
                'alamat'  => $this->alamat_instansi_kedua,
                'mulai'   => $this->tanggal_mulai_kedua,
                'selesai' => $this->tanggal_selesai_kedua,
                'nomor'   => $this->nomor_surat_kedua,
            ];
        }

        return $daftar;
    }

    /** Path PDF form pengajuan (yang ditandatangani mahasiswa) untuk instansi ke-$urutan. */
    public function fileSuratPengajuan(int $urutan = 1): ?string
    {
        return match ($urutan) {
            1       => $this->file_surat_pengajuan,
            2       => $this->file_surat_pengajuan_kedua,
            default => null,
        };
    }

    /** Path PDF surat pengantar magang untuk instansi ke-$urutan. */
    public function fileSuratMagang(int $urutan = 1): ?string
    {
        return match ($urutan) {
            1       => $this->file_surat_magang,
            2       => $this->file_surat_magang_kedua,
            default => null,
        };
    }

    /* =========================================================
       MAHASISWA: ketua selalu dari akun (users), anggota dari kolom "anggota"
    ========================================================= */

    /**
     * Anggota kelompok SELAIN ketua.
     *
     * Ketua tidak diambil dari kolom "anggota", melainkan selalu dari akun (users),
     * supaya nama / NIM ketua tidak pernah usang atau dobel setelah diubah admin.
     * Entri ketua di kolom "anggota" (snapshot lama) dikenali dari "user_id" yang terisi
     * atau NIM yang sama dengan ketua, lalu dilewati. NIM kembar juga dilewati.
     *
     * @return array<int, array{nama: string, nim: string, departemen: ?string}>
     */
    public function anggotaTambahan(): array
    {
        $this->loadMissing('user');

        $terpakai = [strtoupper(trim((string) $this->user?->nim)) => true];
        $hasil = [];

        foreach ($this->anggota ?? [] as $a) {
            $nim = strtoupper(trim((string) ($a['nim'] ?? '')));

            if ($nim === '' || ! empty($a['user_id']) || isset($terpakai[$nim])) {
                continue;
            }

            $terpakai[$nim] = true;

            $hasil[] = [
                'nama'       => (string) ($a['nama'] ?? ''),
                'nim'        => $nim,
                'departemen' => $a['departemen'] ?? null, // kode departemen
            ];
        }

        return $hasil;
    }

    /** Jumlah mahasiswa dalam pengajuan, termasuk ketua. */
    public function jumlahMahasiswa(): int
    {
        return 1 + count($this->anggotaTambahan());
    }

    /**
     * Daftar mahasiswa yang dicetak di surat: pengaju (ketua) selalu nomor 1 dengan data
     * terbaru dari akun, lalu anggota tambahan. Departemen diubah dari kode menjadi nama.
     * Total maksimal config('magang.maks_anggota').
     */
    public function daftarAnggotaCetak(): array
    {
        $this->loadMissing('user');
        $user = $this->user;

        $daftar = [[
            'nama'       => $user->name,
            'nim'        => $user->nim,
            'departemen' => Departemen::nama($user->departemen),
        ]];

        foreach ($this->anggotaTambahan() as $a) {
            $daftar[] = [
                'nama'       => $a['nama'],
                'nim'        => $a['nim'],
                'departemen' => Departemen::nama($a['departemen']),
            ];
        }

        return array_slice($daftar, 0, (int) config('magang.maks_anggota', 10));
    }
}

<?php

namespace App\Http\Controllers;

use App\Enums\StatusPengajuan;
use App\Models\PengajuanMagang;
use Illuminate\Support\Facades\Storage;
use Illuminate\Support\Str;

class UnduhSuratMagangController extends Controller
{
    public function __invoke(int $pengajuan, int $urutan)
    {
        $p = PengajuanMagang::find($pengajuan);

        abort_if(
            $p === null
            || $p->status !== StatusPengajuan::Setuju
            || $urutan < 1
            || $urutan > $p->jumlahInstansi(),
            404,
            'Surat tidak tersedia.'
        );

        $path = $p->fileSuratMagang($urutan);

        abort_if(blank($path), 404, 'Surat tidak tersedia.');

        // Hanya disk privat. Folder storage tidak pernah dibuka ke publik.
        $disk = Storage::disk('local');

        abort_unless($disk->exists($path), 404, 'Surat tidak tersedia.');

        $namaInstansi = $urutan === 2 ? $p->instansi_kedua : $p->instansi;
        $slug = Str::limit(Str::slug((string) $namaInstansi), 60, '');
        $namaFile = "surat-magang-{$urutan}" . ($slug !== '' ? "-{$slug}" : '') . '.pdf';

        return $disk->download($path, $namaFile, [
            'Content-Type'           => 'application/pdf',
            'Cache-Control'          => 'private, no-store',
            'X-Content-Type-Options' => 'nosniff',
            'X-Robots-Tag'           => 'noindex, nofollow',
        ]);
    }
}

<?php

namespace App\Http\Controllers;

use App\Models\PengajuanMagang;
use Illuminate\Filesystem\FilesystemAdapter;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Storage;
use App\Support\Departemen;

class VerifikasiSuratController extends Controller
{
    /**
     * Dituju oleh QR. Menampilkan PDF surat pengajuan itu sendiri. Publik, tanpa login.
     * Surat instansi kedua memakai ?urutan=2 (QR-nya sudah membawa parameter itu).
     */
    public function surat(Request $request, string $kode)
    {
        $pengajuan = PengajuanMagang::where('kode_verifikasi', $kode)
            ->whereNotNull('ditandatangani_at')
            ->first();

        $file = $pengajuan?->fileSuratPengajuan((int) $request->query('urutan', 1));

        abort_if(
            ! $pengajuan || ! $file,
            404,
            'Surat tidak ditemukan atau tanda tangan sudah dibatalkan.'
        );

        /** @var FilesystemAdapter $disk */
        $disk = Storage::disk('local');

        abort_if(! $disk->exists($file), 404, 'File surat tidak ditemukan.');

        // tampil langsung di browser (inline)
        return $disk->response($file, 'surat-pengajuan.pdf', [
            'Content-Type' => 'application/pdf',
        ]);
    }

    /** Versi data (JSON) untuk kebutuhan aplikasi, datanya sengaja minimal. */
    public function show(Request $request, string $kode)
    {
        $pengajuan = PengajuanMagang::with('user')
            ->where('kode_verifikasi', $kode)
            ->whereNotNull('ditandatangani_at')
            ->first();

        $urutan   = (int) $request->query('urutan', 1);
        $instansi = $pengajuan?->daftarInstansi()[$urutan - 1] ?? null;

        if (! $pengajuan || ! $instansi) {
            return response()->json([
                'valid'   => false,
                'message' => 'Surat tidak ditemukan atau tidak valid.',
            ], 404);
        }

        return response()->json([
            'valid' => true,
            'data'  => [
                'nama'       => $pengajuan->user?->name ?? $pengajuan->nama,
                'nim'        => $pengajuan->user?->nim,
                'departemen' => Departemen::nama($pengajuan->user?->departemen),
                'instansi'               => $instansi['nama'],
                'tanggal_ditandatangani' => $pengajuan->ditandatangani_at?->toDateString(),
                'status'                 => $pengajuan->status?->label(),
            ],
        ]);
    }
}

<?php

namespace App\Http\Controllers\Admin;

use App\Enums\StatusPengajuan;
use App\Http\Controllers\Controller;
use App\Models\PengajuanMagang;
use Illuminate\Filesystem\FilesystemAdapter;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Storage;

class BerkasPengajuanController extends Controller
{
    /** jenis di URL => kolom di tabel */
    private const BERKAS = [
        'surat-pengajuan'        => 'file_surat_pengajuan',
        'surat-pengajuan-kedua'  => 'file_surat_pengajuan_kedua',
        'bukti-spp'              => 'file_bukti_spp',
        'surat-rekomendasi'      => 'file_surat_rekomendasi',
        'surat-magang'           => 'file_surat_magang',
        'surat-magang-kedua'     => 'file_surat_magang_kedua',
    ];

    private const ALIAS = [
        'spp'         => 'bukti-spp',
        'rekomendasi' => 'surat-rekomendasi',
    ];

    /** Berkas yang boleh dibuka walau pengajuan masih draft (surat yang sudah ditandatangani). */
    private const BOLEH_SAAT_DRAFT = ['surat-pengajuan', 'surat-pengajuan-kedua'];

    public function tampil(Request $request, PengajuanMagang $pengajuan, string $jenis)
    {
        $mode = $request->query('mode', 'preview');

        // Kompatibel dengan link bentuk lama: /berkas/download?nama=surat-magang
        if (in_array($jenis, ['preview', 'download'], true)) {
            $mode  = $jenis;
            $jenis = (string) $request->query('nama', $request->query('jenis', ''));
        }

        $jenis = self::ALIAS[$jenis] ?? $jenis;

        abort_unless(isset(self::BERKAS[$jenis]), 404, "Jenis berkas tidak dikenal: {$jenis}");
        abort_if(
            $pengajuan->status === StatusPengajuan::Draft && ! in_array($jenis, self::BOLEH_SAAT_DRAFT, true),
            404,
            'Pengajuan belum dikirim.'
        );

        $path = $pengajuan->{self::BERKAS[$jenis]};

        abort_if(! $path, 404, 'Berkas belum tersedia.');

        $lokasi = $this->cariFile($path);

        abort_if(! $lokasi, 404, "File tidak ditemukan di penyimpanan: {$path}");

        [$disk, $pathAsli] = $lokasi;

        $ekstensi = pathinfo($pathAsli, PATHINFO_EXTENSION);
        $nama     = "{$jenis}-{$pengajuan->id}.{$ekstensi}";

        return $mode === 'download'
            ? $disk->download($pathAsli, $nama)
            : $disk->response($pathAsli, $nama);   // tampil langsung di browser (inline)
    }

    /**
     * Mencari file di disk privat dulu, lalu disk public (untuk data lama).
     * Awalan "public/" atau "storage/" pada path lama ikut dicoba tanpa awalan itu.
     *
     * @return array{0: FilesystemAdapter, 1: string}|null
     */
    private function cariFile(string $path): ?array
    {
        $kandidat = array_values(array_unique([
            $path,
            ltrim((string) preg_replace('#^(public/|storage/)#', '', $path), '/'),
        ]));

        foreach (['local', 'public'] as $namaDisk) {
            /** @var FilesystemAdapter $disk */
            $disk = Storage::disk($namaDisk);

            foreach ($kandidat as $p) {
                if ($disk->exists($p)) {
                    return [$disk, $p];
                }
            }
        }

        return null;
    }
}

<?php

namespace App\Http\Controllers\Admin;

use App\Enums\StatusPengajuan;
use App\Http\Controllers\Controller;
use App\Http\Requests\UbahStatusPengajuanRequest;
use App\Http\Requests\Admin\UpdatePengajuanMagangRequest;
use App\Http\Resources\PengajuanMagangAdminDetailResource;
use App\Http\Resources\PengajuanMagangAdminResource;
use App\Models\PengajuanMagang;
use App\Models\User;
use App\Services\PengajuanMagangService;
use App\Services\VerifikasiPengajuanService;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Illuminate\Validation\Rule;
use App\Support\Departemen;

class PengajuanMagangController extends Controller
{
    public function __construct(
        private VerifikasiPengajuanService $verifikasi,
        private PengajuanMagangService $pengajuanService
    ) {}

    /**
     * Halaman tabel: pengajuan yang sudah dikirim (tanpa draft), 20 per halaman.
     * Urutan: nomor surat naik (1, 2, 3, ...), yang belum bernomor di bawah.
     */
    public function index(Request $request)
    {
        $filter = $request->validate([
            'status'     => ['nullable', Rule::in(['menunggu_verifikasi', 'setuju', 'revisi'])],
            'q'          => ['nullable', 'string', 'max:100'],
            'departemen' => ['nullable', 'string', 'max:100'],
            'per_page'   => ['nullable', 'integer', 'min:1', 'max:100'],
        ]);

        $daftar = $this->urutkanNomorSurat($this->kueri($filter)->with('user'))
            ->paginate($filter['per_page'] ?? 20)
            ->withQueryString();

        return PengajuanMagangAdminResource::collection($daftar);
    }

    /**
     * Jumlah per status (mengikuti pencarian dan departemen yang aktif)
     * dan daftar departemen untuk dropdown.
     */
    public function ringkasan(Request $request)
    {
        $filter = $request->validate([
            'q'          => ['nullable', 'string', 'max:100'],
            'departemen' => ['nullable', 'string', 'max:100'],
        ]);

        $hitung = $this->kueri($filter, denganStatus: false)
            ->selectRaw('status, count(*) as total')
            ->groupBy('status')
            ->pluck('total', 'status');

        // SESUAIKAN nama kolom departemen di tabel users
        $departemen = User::whereIn(
            'id',
            PengajuanMagang::where('status', '!=', StatusPengajuan::Draft->value)->select('user_id')
        )
            ->whereNotNull('departemen')
            ->distinct()
            ->pluck('departemen')
            ->map(fn($kode) => ['kode' => $kode, 'nama' => Departemen::nama($kode)])
            ->sortBy('nama')
            ->values();

        return response()->json([
            'data' => [
                'menunggu_verifikasi' => (int) ($hitung['menunggu_verifikasi'] ?? 0),
                'setuju'              => (int) ($hitung['setuju'] ?? 0),
                'revisi'              => (int) ($hitung['revisi'] ?? 0),
                'departemen'          => $departemen,
            ],
        ]);
    }

    /** Tombol "Detail": data lengkap dan daftar berkas. */
    public function show(PengajuanMagang $pengajuan)
    {
        abort_if($pengajuan->status === StatusPengajuan::Draft, 404);

        return new PengajuanMagangAdminDetailResource($pengajuan->load('user'));
    }

    /**
     * Admin mengedit data pengajuan magang (termasuk data mahasiswa pengaju,
     * anggota kelompok, instansi 1, dan instansi 2).
     */
    public function update(
        UpdatePengajuanMagangRequest $request,
        PengajuanMagang $pengajuan
    ) {
        // Draft tidak bisa diedit melalui halaman admin.
        abort_if(
            $pengajuan->status === StatusPengajuan::Draft,
            404
        );

        // Pengajuan yang sudah disetujui tidak boleh diedit
        // karena surat dan tanda tangan mungkin sudah diterbitkan.
        abort_if(
            !in_array(
                $pengajuan->status->value,
                ['menunggu_verifikasi', 'revisi'],
                true
            ),
            403,
            'Pengajuan yang sudah disetujui tidak dapat diedit.'
        );

        $pengajuan = $this->pengajuanService->perbaruiOlehAdmin(
            $pengajuan,
            $request->validated()
        );

        return (new PengajuanMagangAdminDetailResource(
            $pengajuan->load('user')
        ))->additional([
            'message' => 'Data pengajuan magang berhasil diperbarui.',
        ]);
    }

    /** Admin memutuskan: setuju atau revisi (catatan wajib saat revisi). */
    public function ubahStatus(UbahStatusPengajuanRequest $request, PengajuanMagang $pengajuan)
    {
        abort_if($pengajuan->status === StatusPengajuan::Draft, 404);

        if ($request->validated('status') === 'setuju') {
            $pengajuan = $this->verifikasi->setujui($pengajuan, $request->user());
            $pesan = $pengajuan->jumlahInstansi() === 2
                ? 'Pengajuan disetujui dan 2 surat pengantar magang telah dibuat.'
                : 'Pengajuan disetujui dan surat pengantar magang telah dibuat.';
        } else {
            $pengajuan = $this->verifikasi->mintaRevisi(
                $pengajuan,
                $request->user(),
                $request->validated('catatan'),
            );
            $pesan = 'Pengajuan dikembalikan ke mahasiswa untuk direvisi.';
        }

        return (new PengajuanMagangAdminDetailResource($pengajuan->load('user')))
            ->additional(['message' => $pesan]);
    }

    /**
     * Urutkan berdasarkan tahun surat (descending) lalu angka urut di depan nomor surat.
     * Format: "8/IT3.F4/PK.01.06/2026" -> urut=8, tahun=2026.
     * Angka dibaca numerik supaya 10 tidak muncul sebelum 2.
     * Pengajuan tanpa nomor surat ditaruh paling bawah, urut berdasarkan id.
     */
    private function urutkanNomorSurat($query)
    {
        $angka = match (DB::connection()->getDriverName()) {
            'pgsql'  => "CAST(split_part(nomor_surat, '/', 1) AS INTEGER)",
            'sqlite' => "CAST(substr(nomor_surat, 1, instr(nomor_surat, '/') - 1) AS INTEGER)",
            default  => "CAST(SUBSTRING_INDEX(nomor_surat, '/', 1) AS UNSIGNED)", // MySQL / MariaDB
        };

        $tahun = match (DB::connection()->getDriverName()) {
            'pgsql'  => "CAST(split_part(nomor_surat, '/', 4) AS INTEGER)",
            'sqlite' => "CAST(substr(nomor_surat, -4) AS INTEGER)", // 4 digit tahun di akhir nomor surat
            default  => "CAST(SUBSTRING_INDEX(nomor_surat, '/', -1) AS UNSIGNED)", // MySQL / MariaDB
        };

        return $query
            ->orderByRaw("CASE WHEN nomor_surat IS NULL OR nomor_surat = '' THEN 1 ELSE 0 END")
            ->orderByRaw("{$tahun} DESC")
            ->orderByRaw("{$angka} ASC")
            ->orderBy('id');
    }

    /** Kueri dasar yang dipakai daftar dan ringkasan, supaya hasilnya selalu sejalan. */
    private function kueri(array $filter, bool $denganStatus = true)
    {
        return PengajuanMagang::query()
            ->where('status', '!=', StatusPengajuan::Draft->value)
            ->when(
                $denganStatus ? ($filter['status'] ?? null) : null,
                fn($query, $status) => $query->where('status', $status)
            )
            ->when(
                $filter['departemen'] ?? null,
                // SESUAIKAN nama kolom departemen di tabel users
                fn($query, $dep) => $query->whereHas('user', fn($user) => $user->where('departemen', $dep))
            )
            ->when($filter['q'] ?? null, function ($query, $kata) {
                $query->where(function ($query) use ($kata) {
                    $query->where('instansi', 'like', "%{$kata}%")
                        ->orWhere('instansi_kedua', 'like', "%{$kata}%")
                        ->orWhere('nomor_surat', 'like', "%{$kata}%")
                        ->orWhere('nomor_surat_kedua', 'like', "%{$kata}%")
                        // SESUAIKAN nama kolom users (name, nim)
                        ->orWhereHas('user', fn($user) => $user
                            ->where('name', 'like', "%{$kata}%")
                            ->orWhere('nim', 'like', "%{$kata}%"));
                });
            });
    }

class AuthController extends Controller

    /**
     * Registrasi mahasiswa baru.
     */
    public function register(RegisterRequest $request): JsonResponse
    {
        $data = $request->validated();

        $data['name'] = preg_replace_callback(
        "/(^|[\s-])([\p{L}])/u",
        fn ($matches) => $matches[1] . mb_strtoupper($matches[2]),
        mb_strtolower($data['name'])
);

        // Departemen ditentukan backend berdasarkan prefix NIM.
        $departemen = Departemen::dariNim($data['nim']);

        if (! $departemen) {
            throw ValidationException::withMessages([
                'nim' => 'Departemen untuk NIM ini tidak dikenali.',
            ]);
        }

        $user = DB::transaction(function () use ($data, $departemen) {
            // Buat akun mahasiswa.
            $user = User::create([
                'name' => $data['name'],
                'nim' => $data['nim'],
                'departemen' => $departemen['kode'],
                'email' => $data['email'],
                'no_wa' => $data['no_wa'],
                'password' => $data['password'],
                'role' => 'mahasiswa',
            ]);

            // Ambil atau buat role mahasiswa dengan guard yang sesuai.
            $role = Role::findOrCreate('mahasiswa', 'web');

            // Hubungkan role Spatie secara otomatis.
            $user->assignRole($role);

            return $user;
        });

        return response()->json([
            'message' => 'Registrasi berhasil. Silakan login.',
            'user' => new UserResource($user),
        ], 201);
    }

    /**
     * Login pengguna.
     */
    public function login(LoginRequest $request): JsonResponse
    {
        $data = $request->validated();

        $user = User::where('email', $data['email'])->first();

        // Gunakan pesan yang sama untuk email atau password yang salah.
        if (
            ! $user ||
            ! Hash::check($data['password'], $user->password)
        ) {
            return response()->json([
                'message' => 'Email atau password salah.',
            ], 401);
        }

        // Buat token autentikasi Sanctum.
        $token = $user->createToken('auth_token')->plainTextToken;

        return response()->json([
            'message' => 'Login berhasil.',
            'token' => $token,
            'token_type' => 'Bearer',
            'user' => new UserResource($user),
        ]);
    }

    /**
     * Mendapatkan informasi pengguna yang sedang login.
     */
    public function me(Request $request): UserResource
    {
        return new UserResource($request->user());
    }

    /**
     * Logout dan mencabut token yang sedang digunakan.
     */
    public function logout(Request $request): JsonResponse
    {
        $request->user()->currentAccessToken()->delete();

        return response()->json([
            'message' => 'Logout berhasil.',
        ]);
    }

<?php

namespace App\Services;

use Illuminate\Support\Facades\DB;

class NomorSuratService
{
    /**
     * Membuat nomor surat berikutnya, aman dari bentrok (baris dikunci selama transaksi).
     * Dipanggil di dalam transaksi persetujuan, jadi nomor tidak "terbuang" kalau proses gagal.
     */
    public function berikutnya(?int $tahun = null): string
    {
        $tahun ??= now()->year;
        $awal = max((int) config('magang.nomor_surat.mulai_dari', 1), 1);

        return DB::transaction(function () use ($tahun, $awal) {
            DB::table('nomor_surat_urut')->insertOrIgnore([
                'tahun'    => $tahun,
                'terakhir' => $awal - 1,
            ]);

            $baris = DB::table('nomor_surat_urut')
                ->where('tahun', $tahun)
                ->lockForUpdate()
                ->first();

            $urut = $baris->terakhir + 1;

            DB::table('nomor_surat_urut')
                ->where('tahun', $tahun)
                ->update(['terakhir' => $urut]);

            return str_replace(
                ['{urut}', '{tahun}'],
                [$urut, $tahun],
                config('magang.nomor_surat.format'),
            );
        });
    }
}

    public function buatDraft(User $user, array $data): PengajuanMagang
    {
        $pengajuan = new PengajuanMagang();

        $pengajuan->instansi = $this->formatInstansi($data['instansi']);
        $pengajuan->alamat_instansi = $this->formatInstansi($data['alamat_instansi']);
        $pengajuan->tanggal_mulai = $data['tanggal_mulai'];
        $pengajuan->tanggal_selesai = $data['tanggal_selesai'];

        // Instansi kedua (opsional) -> surat kedua.
        $this->isiInstansiKedua($pengajuan, $data);

        $pengajuan->user_id = $user->id;
        $pengajuan->nama = $user->name;
        $pengajuan->anggota = $this->susunAnggota($user, $data['anggota'] ?? []);
        $pengajuan->status = StatusPengajuan::Draft;
        $pengajuan->kode_verifikasi = Str::random(32);

        $pengajuan->save();

        return $pengajuan;
    }

    public function perbarui(PengajuanMagang $pengajuan, array $data): PengajuanMagang
    {
        $this->pastikanBisaDiedit($pengajuan);
        $pengajuan->loadMissing('user');

        // Surat final instansi 1 dan 2 (bila ada) akan dibatalkan.
        $fileLama = [
            $pengajuan->file_surat_pengajuan,
            $pengajuan->file_surat_pengajuan_kedua,
        ];

        $pengajuan->instansi = $this->formatInstansi($data['instansi']);
        $pengajuan->alamat_instansi = $this->formatInstansi($data['alamat_instansi']);
        $pengajuan->tanggal_mulai = $data['tanggal_mulai'];
        $pengajuan->tanggal_selesai = $data['tanggal_selesai'];

        // Instansi kedua hanya disentuh jika dikirim (null/kosong berarti dihapus).
        if (array_key_exists('instansi_kedua', $data)) {
            $this->isiInstansiKedua($pengajuan, $data);
        }

        // Ketua selalu mengikuti akun (bisa saja sudah diubah admin), anggota tambahan
        // diganti bila dikirim dan dipertahankan bila tidak.
        $pengajuan->nama = $pengajuan->user->name;
        $pengajuan->anggota = $this->anggotaBaru($pengajuan, $data);

        // Batalkan tanda tangan sebelumnya.
        $pengajuan->ditandatangani_at = null;
        $pengajuan->file_surat_pengajuan = null;
        $pengajuan->file_surat_pengajuan_kedua = null;

        $pengajuan->save();

        $this->hapusFiles($fileLama);

        return $pengajuan->refresh();
    }

     * Tahap 2: preview surat untuk instansi ke-$urutan (1 atau 2).
     * Kalau sudah ditandatangani, tampilkan versi final.

    public function previewSurat(PengajuanMagang $pengajuan, int $urutan = 1): string
    {
        abort_if(
            $urutan < 1 || $urutan > $pengajuan->jumlahInstansi(),
            404,
            'Surat untuk instansi tersebut tidak ada.'
        );

        if ($pengajuan->sudahDitandatangani()) {
            $path = $pengajuan->fileSuratPengajuan($urutan);
            $disk = Storage::disk('local');

            // Pastikan file surat yang ditandatangani masih tersedia.
            abort_unless(
                $path && $disk->exists($path),
                404,
                'File surat yang ditandatangani tidak ditemukan.'
            );

            return $disk->get($path);
        }

        // Jika belum ditandatangani, tampilkan preview tanpa QR.
        return $this->surat->render($pengajuan, denganQr: false, urutan: $urutan);
    }

    /**
     * Tahap 2b: tanda tangan. Surat final dengan QR dibuat dan disimpan,
     * satu PDF untuk tiap instansi (1 atau 2 surat).
     */
    public function tandaTangani(PengajuanMagang $pengajuan): PengajuanMagang
    {
        $this->pastikanBisaDiedit($pengajuan);

        if ($pengajuan->sudahDitandatangani()) {
            return $pengajuan;
        }

        $pathBaru = [];

        try {
            // Atur timestamp agar data surat yang dirender sesuai.
            $pengajuan->ditandatangani_at = now();

            foreach ($pengajuan->daftarInstansi() as $instansi) {
                $path = $this->surat->simpan($pengajuan, $instansi['urutan']);

                if (! $path) {
                    throw new \RuntimeException('Gagal menyimpan surat pengajuan.');
                }

                $pathBaru[$instansi['urutan']] = $path;
            }

            $pengajuan->file_surat_pengajuan = $pathBaru[1];
            $pengajuan->file_surat_pengajuan_kedua = $pathBaru[2] ?? null;
            $pengajuan->save();

            return $pengajuan->refresh();
        } catch (Throwable $e) {
            // Bersihkan file baru jika proses gagal.
            $this->hapusFiles($pathBaru);

            throw $e;
        }
    }

    /** Membatalkan tanda tangan: semua surat final dan QR dihapus, preview kembali tanpa QR. */
    public function batalkanTandaTangan(PengajuanMagang $pengajuan): PengajuanMagang
    {
        $this->pastikanBisaDiedit($pengajuan);

        if (! $pengajuan->sudahDitandatangani()) {
            return $pengajuan;

        $fileLama = [
            $pengajuan->file_surat_pengajuan,
            $pengajuan->file_surat_pengajuan_kedua,
        ];

        // Hapus referensi tanda tangan dari database.
        $pengajuan->ditandatangani_at = null;
        $pengajuan->file_surat_pengajuan = null;
        $pengajuan->file_surat_pengajuan_kedua = null;
        $pengajuan->save();

        // Hapus PDF lama setelah database berhasil disimpan.
        $this->hapusFiles($fileLama);

        return $pengajuan->refresh();
    }

    public function kirim(
        PengajuanMagang $pengajuan,
        ?UploadedFile $buktiSpp,
        ?UploadedFile $suratRekomendasi
    ): PengajuanMagang {
        $this->pastikanBisaDiedit($pengajuan);

        if (! $pengajuan->sudahDitandatangani()) {
            throw ValidationException::withMessages([
                'tanda_tangan' => 'Surat pengajuan harus ditandatangani terlebih dahulu.',
            ]);
        }

        if (
            (! $buktiSpp && ! $pengajuan->file_bukti_spp) ||
            (! $suratRekomendasi && ! $pengajuan->file_surat_rekomendasi)
        ) {
            throw ValidationException::withMessages([
                'berkas' => 'Bukti SPP dan surat rekomendasi wajib tersedia.',
            ]);
        }

        $folder = "pengajuan/{$pengajuan->id}";
        $berkasBaru = [];
        $berkasLama = [
            'file_bukti_spp'         => $pengajuan->file_bukti_spp,
            'file_surat_rekomendasi' => $pengajuan->file_surat_rekomendasi,
        ];

        try {
            // Simpan berkas baru tanpa menghapus berkas lama.
            if ($buktiSpp) {
                $berkasBaru['file_bukti_spp'] = $this->simpanBerkas($buktiSpp, $folder, 'bukti SPP');
            }

            if ($suratRekomendasi) {
                $berkasBaru['file_surat_rekomendasi'] = $this->simpanBerkas($suratRekomendasi, $folder, 'surat rekomendasi');
            }

            // Perbarui database dalam transaksi.
            DB::transaction(function () use ($pengajuan, $berkasBaru) {
                foreach ($berkasBaru as $kolom => $path) {
                    $pengajuan->{$kolom} = $path;
                }

                $pengajuan->status = StatusPengajuan::MenungguVerifikasi;
                $pengajuan->dikirim_at = now();
                $pengajuan->save();
            });
        } catch (Throwable $e) {
            // Jika proses gagal, bersihkan berkas baru yang sempat tersimpan.
            $this->hapusFiles($berkasBaru);

            throw $e;
        }

        // Hapus berkas lama hanya setelah database berhasil diperbarui.
        foreach ($berkasBaru as $kolom => $path) {
            $pathLama = $berkasLama[$kolom] ?? null;

            if ($pathLama && $pathLama !== $path) {
                $this->hapusFile($pathLama);
            }
        }

        return $pengajuan->refresh();
    }

    /**
     * Memperbarui data pengajuan magang oleh admin (termasuk data mahasiswa pengaju,
     * anggota kelompok, instansi 1, dan instansi 2).
     *
     * Ketua dan akun mahasiswa adalah satu data: nama / NIM / departemen yang diubah di sini
     * langsung disimpan ke akun (users), dan ketua di surat selalu dibaca dari akun.
     *
     * @param array<string, mixed> $data
     * @throws ValidationException
     */
    public function perbaruiOlehAdmin(PengajuanMagang $pengajuan, array $data): PengajuanMagang
    {
        $pengajuan->loadMissing('user');

        $fileBaru = [];
        $fileLama = [];

        try {
            DB::transaction(function () use ($pengajuan, $data, &$fileBaru, &$fileLama): void {
                $user = $pengajuan->user;

                // 1. Data mahasiswa pengaju (akun)
                if (isset($data['nama'])) {
                    $user->name = $this->formatNama($data['nama']);
                }

                if (isset($data['nim'])) {
                    $nim = strtoupper(trim($data['nim']));
                    $departemen = Departemen::dariNim($nim);

                    if (! $departemen) {
                        throw ValidationException::withMessages([
                            'nim' => 'Departemen untuk NIM ini tidak dikenali.',
                        ]);
                    }

                    $user->nim = $nim;
                    $user->departemen = $departemen['kode'];
                }

                if ($user->isDirty()) {
                    $user->save();
                }

                $pengajuan->nama = $user->name;

                // 2. Instansi pertama
                if (isset($data['instansi'])) {
                    $pengajuan->instansi = $this->formatInstansi($data['instansi']);
                }

                if (isset($data['alamat_instansi'])) {
                    $pengajuan->alamat_instansi = $this->formatInstansi($data['alamat_instansi']);
                }

                foreach (['tanggal_mulai', 'tanggal_selesai'] as $kolom) {
                    if (isset($data[$kolom])) {
                        $pengajuan->{$kolom} = $data[$kolom];
                    }
                }

                // 3. Instansi kedua (kosong berarti dihapus beserta surat finalnya)
                if (array_key_exists('instansi_kedua', $data)) {
                    $this->isiInstansiKedua($pengajuan, $data);

                    if ($pengajuan->jumlahInstansi() === 1 && $pengajuan->file_surat_pengajuan_kedua) {
                        $fileLama[] = $pengajuan->file_surat_pengajuan_kedua;
                        $pengajuan->file_surat_pengajuan_kedua = null;
                    }
                }

                // 4. Anggota: ketua dibangun ulang dari akun, jadi tidak ada data ketua lama yang tersisa.
                $pengajuan->anggota = $this->anggotaBaru($pengajuan, $data);

                $pengajuan->save();

                // 5. Surat yang sudah ditandatangani dirender ulang dengan data terbaru.
                if ($pengajuan->sudahDitandatangani()) {
                    foreach ([1 => 'file_surat_pengajuan', 2 => 'file_surat_pengajuan_kedua'] as $urutan => $kolom) {
                        if ($urutan > $pengajuan->jumlahInstansi()) {
                            continue;
                        }

                        if ($pengajuan->{$kolom}) {
                            $fileLama[] = $pengajuan->{$kolom};
                        }

                        $path = $this->surat->simpan($pengajuan, $urutan);

                        if (! $path) {
                            throw new \RuntimeException('Gagal menyimpan ulang surat pengajuan.');
                        }

                        $fileBaru[] = $path;
                        $pengajuan->{$kolom} = $path;
                    }

                    $pengajuan->save();
                }
            });
        } catch (Throwable $e) {
            // Transaksi dibatalkan: buang surat baru yang sempat dibuat.
            $this->hapusFiles($fileBaru);

            throw $e;
        }

        // Surat lama dihapus setelah database berhasil disimpan
        // (path yang dipakai ulang oleh surat baru tidak ikut dihapus).
        $this->hapusFiles(array_diff($fileLama, $fileBaru));

        return $pengajuan->refresh();
    }

       HELPER
    ========================================================= */

    /**
     * Instansi kedua: terisi bila nama instansi dikirim, selain itu semua kolomnya dikosongkan.
     */
    private function isiInstansiKedua(PengajuanMagang $pengajuan, array $data): void
    {
        if (filled($data['instansi_kedua'] ?? null)) {
            $pengajuan->instansi_kedua = $this->formatInstansi($data['instansi_kedua']);
            $pengajuan->alamat_instansi_kedua = $this->formatInstansi($data['alamat_instansi_kedua']);
            $pengajuan->tanggal_mulai_kedua = $data['tanggal_mulai_kedua'];
            $pengajuan->tanggal_selesai_kedua = $data['tanggal_selesai_kedua'];

            return;
        }

        $pengajuan->instansi_kedua = null;
        $pengajuan->alamat_instansi_kedua = null;
        $pengajuan->tanggal_mulai_kedua = null;
        $pengajuan->tanggal_selesai_kedua = null;
    }

    /**
     * Daftar anggota yang disimpan: anggota tambahan dari $data bila dikirim,
     * selain itu anggota tambahan yang sudah ada. Ketua selalu disusun ulang dari akun.
     */
    private function anggotaBaru(PengajuanMagang $pengajuan, array $data): array
    {
        $tambahan = array_key_exists('anggota', $data)
            ? ($data['anggota'] ?? [])
            : $pengajuan->anggotaTambahan();

        return $this->susunAnggota($pengajuan->user, $tambahan);
    }

    /**
     * Menyusun daftar kelompok: ketua (dari akun) selalu di urutan pertama, lalu anggota tambahan.
     * Entri yang NIM-nya sama dengan ketua, kembar, atau berasal dari snapshot ketua lama
     * (punya user_id) dilewati begitu saja. Total dibatasi config('magang.maks_anggota').
     */
    private function susunAnggota(User $user, ?array $anggotaTambahan): array
    {
        $nimKetua = strtoupper(trim((string) $user->nim));

        $anggota = [
            [
                'user_id'    => $user->id,
                'nama'       => $user->name,
                'nim'        => $user->nim,
                'departemen' => $user->departemen,
            ],
        ];

        $terpakai = [$nimKetua => true];

        foreach ($anggotaTambahan ?? [] as $member) {
            $nim = strtoupper(trim((string) ($member['nim'] ?? '')));

            if ($nim === '' || ! empty($member['user_id']) || isset($terpakai[$nim])) {
                continue;
            }

            // Ambil departemen otomatis berdasarkan NIM
            $departemen = Departemen::dariNim($nim);

            if (! $departemen) {
                throw ValidationException::withMessages([
                    'anggota' => "NIM {$nim} memiliki kode departemen yang tidak dikenal.",
                ]);
            }

            $terpakai[$nim] = true;

            $anggota[] = [
                'user_id'    => null,
                'nama'       => $this->formatNama((string) ($member['nama'] ?? '')),
                'nim'        => $nim,
                'departemen' => $departemen['kode'],
            ];
        }

        $maks = (int) config('magang.maks_anggota', 10);

        if (count($anggota) > $maks) {
            throw ValidationException::withMessages([
                'anggota' => "Maksimal {$maks} mahasiswa per pengajuan (termasuk ketua).",
            ]);
        }

        return $anggota;
    }

    /**
     * Nama instansi / alamat: tiap kata diawali huruf kapital, dan singkatan
     * (PT, CV, KPH, KM, RT, dst.) ditulis kapital penuh.
     */
    private function formatInstansi(string $value): string
    {
        $value = $this->formatNama($value);

        $singkatan = array_unique(array_map(
            'mb_strtolower',
            array_merge(self::SINGKATAN, (array) config('magang.singkatan_instansi', []))
        ));

        $pola = '/\b(' . implode('|', array_map(
            fn ($kata) => preg_quote($kata, '/'),
            $singkatan
        )) . ')\b/iu';

        return preg_replace_callback(
            $pola,
            fn ($cocok) => mb_strtoupper($cocok[1]),
            $value
        );
    }

    private function formatNama(string $value): string
    {
        $value = mb_strtolower(trim($value));

        return preg_replace_callback(
            "/(^|[\s-])([\p{L}])/u",
            fn ($matches) => $matches[1] . mb_strtoupper($matches[2]),
            $value
        );
    }

    private function simpanBerkas(UploadedFile $file, string $folder, string $nama): string
    {
        $path = $file->store($folder, 'local');

        if (! $path) {
            throw new \RuntimeException("Gagal menyimpan {$nama}.");
        }

        return $path;
    }

    private function pastikanBisaDiedit(PengajuanMagang $pengajuan): void
    {
        if (! $pengajuan->bisaDiedit()) {
            throw ValidationException::withMessages([
                'status' => 'Pengajuan sudah dikirim dan tidak bisa diubah.',
            ]);
        }
    }

    /** @param iterable<int|string, ?string> $paths */
    private function hapusFiles(iterable $paths): void
    {
        foreach ($paths as $path) {
            $this->hapusFile($path);
        }
    }

    private function hapusFile(?string $path): void
    {
        if ($path) {
            Storage::disk('local')->delete($path);
        }
    }
}

namespace App\Services;

use App\Models\PengajuanMagang;
use Dompdf\Dompdf;
use Dompdf\Options;
use Illuminate\Support\Carbon;
use Illuminate\Support\Facades\Storage;

class SuratMagangService
{
    /**
     * Membuat PDF surat pengantar magang untuk instansi ke-$urutan (1 atau 2) dan menyimpannya.
     * Surat pertama memakai nama file lama (surat-magang.pdf) agar data lama tetap cocok.
     */
    public function simpan(PengajuanMagang $pengajuan, int $urutan = 1): string
    {
        $path = $urutan > 1
            ? "pengajuan/{$pengajuan->id}/surat-magang-{$urutan}.pdf"
            : "pengajuan/{$pengajuan->id}/surat-magang.pdf";

        $berhasil = Storage::disk('local')->put(
            $path,
            $this->render($pengajuan, $urutan)
        );

        if (! $berhasil) {
            throw new \RuntimeException(
                'Gagal menyimpan PDF surat pengantar magang.'
            );
        }

        return $path;
    }

    /**
     * Membuat PDF surat pengantar magang untuk instansi ke-$urutan.
     * Isi surat 1 memuat instansi 1, isi surat 2 memuat instansi 2.
     */
    public function render(PengajuanMagang $pengajuan, int $urutan = 1): string
    {
        $pengajuan->loadMissing('user');

        $penandatangan = config('magang.penandatangan');

        $instansi = $pengajuan->daftarInstansi()[$urutan - 1]
            ?? throw new \InvalidArgumentException("Instansi ke-{$urutan} tidak ditemukan.");

        $html = view('surat.suratmagang', [
            'logo' => $this->gambar(config('magang.logo_path')),
            'kop'  => config('magang.kop'),
            'ttd'  => $this->gambar($penandatangan['ttd_path'] ?? null),

            'fakultas'     => config('magang.fakultas'),
            'kota'         => config('magang.kota'),
            'nomor'        => $instansi['nomor'],
            'tanggalSurat' => $this->tanggal(
                $pengajuan->diverifikasi_at ?? now()
            ),

            // Pengaju nomor 1 + anggota tambahan (maks. 10 orang).
            'anggota' => $pengajuan->daftarAnggotaCetak(),

            // Data instansi untuk surat ini.
            'instansi'       => $instansi['nama'],
            'alamat'         => $instansi['alamat'],
            'tanggalMulai'   => $this->tanggal($instansi['mulai']),
            'tanggalSelesai' => $this->tanggal($instansi['selesai']),

            'jabatan'           => $penandatangan['jabatan'],
            'namaPenandatangan' => $penandatangan['nama'],
            'nip'               => $penandatangan['nip'],
        ])->render();

        $options = new Options();
        $options->set('isRemoteEnabled', false);

        $dompdf = new Dompdf($options);
        $dompdf->loadHtml($html);
        $dompdf->setPaper('A4', 'portrait');
        $dompdf->render();

        return $dompdf->output();
    }

    /**
     * Format tanggal berbahasa Indonesia.
     * Contoh: 29 Oktober 2026.
     */
    private function tanggal($nilai): string
    {
        return Carbon::parse($nilai)
            ->locale('id')
            ->translatedFormat('j F Y');
    }

    /**
     * Mengubah gambar di folder public menjadi data URI.
     */
    private function gambar(?string $relatif): ?string
    {
        if (! $relatif) {
            return null;
        }

        $path = public_path($relatif);

        if (! is_file($path)) {
            return null;
        }

        $mime = mime_content_type($path);

        if (! in_array($mime, ['image/png', 'image/jpeg'], true)) {
            return null;
        }

        $konten = file_get_contents($path);

        if ($konten === false) {
            return null;
        }

        return 'data:' . $mime . ';base64,' . base64_encode($konten);
    }
}

        bool $denganQr,
        int $urutan = 1
    ): string {
        $pengajuan->loadMissing('user');

        if (! $pengajuan->user) {
            throw new RuntimeException(
                'Data mahasiswa pemilik pengajuan tidak ditemukan.'
            );
        }

        $instansi = $pengajuan->daftarInstansi()[$urutan - 1]
            ?? throw new RuntimeException("Instansi ke-{$urutan} tidak ditemukan.");

        // Ketua nomor 1 + anggota tambahan. Departemen sudah berupa nama (bukan kode).
        $daftar  = $pengajuan->daftarAnggotaCetak();
        $pengaju = $daftar[0];

        $html = view('surat.pengajuan', [
            'anggota' => $daftar,
            'nama' => $pengaju['nama'],
            'nim' => $pengaju['nim'],
            'departemen' => $pengaju['departemen'],

            'fakultas' => config('magang.fakultas'),
            'kota' => config('magang.kota'),

            'instansi' => $instansi['nama'],
            'alamat' => $instansi['alamat'],

            'tanggalMulai' => $this->tanggal($instansi['mulai']),
            'tanggalSelesai' => $this->tanggal($instansi['selesai']),

            'tanggalSurat' => $this->tanggal(
                $pengajuan->ditandatangani_at ?? now()
            ),

            'qr' => $denganQr ? $this->buatQr($pengajuan, $urutan) : null,
        ])->render();

        $options = new Options();
        $options->set('isRemoteEnabled', false);

        $dompdf = new Dompdf($options);
        $dompdf->loadHtml($html);
        $dompdf->setPaper('A4', 'portrait');
        $dompdf->render();

        return $dompdf->output();
    }

    /** Tanggal berbahasa Indonesia, contoh: "29 Oktober 2026". */
    private function tanggal($nilai): string
    {
        return Carbon::parse($nilai)->locale('id')->translatedFormat('j F Y');
    }

    /**
     * Membuat QR Code yang mengarah ke halaman verifikasi surat.
     * Surat instansi kedua memakai "?urutan=2" supaya QR-nya membuka surat yang tepat.
     */
    private function buatQr(PengajuanMagang $pengajuan, int $urutan = 1): string
    {
        $verifikasiUrl = rtrim((string) config('magang.verifikasi_url'), '/');

        if ($verifikasiUrl === '' || empty($pengajuan->kode_verifikasi)) {
            throw new RuntimeException('URL verifikasi atau kode verifikasi belum dikonfigurasi.');
        }

        $url = $verifikasiUrl . '/' . $pengajuan->kode_verifikasi;

        if ($urutan > 1) {
            $url .= '?urutan=' . $urutan;
        }

        $qrCode = new QrCode(
            data: $url,
            size: 200,
            margin: 5
        );

        return (new PngWriter())
            ->write($qrCode)
            ->getDataUri();
    }

<?php

namespace App\Services;

use App\Enums\StatusPengajuan;
use App\Models\PengajuanMagang;
use Carbon\CarbonInterface;
use Illuminate\Support\Facades\Storage;
use Illuminate\Support\Facades\URL;
use Illuminate\Validation\ValidationException;

class SuratWhatsAppService
{
    /**
     * Membuat link unduh sementara dan URL wa.me untuk satu surat magang.
     * Tidak mengirim apa pun: admin sendiri yang menekan "kirim" di WhatsApp.
     *
     * @return array<string, mixed>
     */
    public function buat(PengajuanMagang $pengajuan, int $urutan): array
    {
        $pengajuan->loadMissing('user');

        if ($pengajuan->status !== StatusPengajuan::Setuju) {
            $this->tolak('status', 'Surat hanya bisa dikirim setelah pengajuan disetujui.');
        }

        $instansi = collect($pengajuan->daftarInstansi())->firstWhere('urutan', $urutan);

        if ($instansi === null) {
            $this->tolak('urutan', "Pengajuan ini tidak memiliki instansi ke-{$urutan}.");
        }

        $path = $pengajuan->fileSuratMagang($urutan);

        if (blank($path)) {
            $this->tolak('urutan', 'Surat magang untuk instansi ini belum tersedia.');
        }

        if (! Storage::disk('local')->exists($path)) {
            $this->tolak('berkas', 'File surat tidak ditemukan di penyimpanan.');
        }

        $noWa = $pengajuan->user?->no_wa;

        if (blank($noWa)) {
            $this->tolak('no_wa', 'Mahasiswa belum memiliki nomor WhatsApp.');
        }

        $nomorTujuan = $this->normalisasiNomor($noWa);

        if ($nomorTujuan === null) {
            $this->tolak('no_wa', 'Format nomor WhatsApp mahasiswa tidak valid.');
        }

        // Masa berlaku link (menit). Default 3 hari; bisa diatur di config/magang.php.
        $menit = max(1, (int) config('magang.link_surat_berlaku_menit', 4320));
        $berlakuSampai = now()->addMinutes($menit);

        $url = URL::temporarySignedRoute('surat-magang.unduh', $berlakuSampai, [
            'pengajuan' => $pengajuan->id,
            'urutan'    => $urutan,
        ]);

        $pesan = $this->susunPesan($pengajuan, $instansi, $url, $berlakuSampai);

        return [
            'urutan'         => $urutan,
            'instansi'       => $instansi['nama'],
            'nomor_surat'    => $instansi['nomor'],
            'nomor_tujuan'   => $nomorTujuan,
            'pesan'          => $pesan,
            'download_url'   => $url,
            'berlaku_sampai' => $berlakuSampai->toIso8601String(),
            'whatsapp_url'   => 'https://wa.me/' . $nomorTujuan . '?text=' . rawurlencode($pesan),
        ];
    }

    /**
     * Mengubah 08xx / +62 8xx / 62-8xx / 8xx menjadi 628xx (format wa.me, tanpa "+").
     * Mengembalikan null bila tidak bisa dikenali sebagai nomor Indonesia.
     */
    public function normalisasiNomor(?string $nomor): ?string
    {
        $digit = preg_replace('/\D+/', '', (string) $nomor); // buang spasi, +, -, kurung

        if ($digit === '') {
            return null;
        }

        if (str_starts_with($digit, '00')) {
            $digit = substr($digit, 2); // 0062... -> 62...
        }

        if (str_starts_with($digit, '62')) {
            // sudah berformat internasional
        } elseif (str_starts_with($digit, '0')) {
            $digit = '62' . substr($digit, 1);
        } elseif (str_starts_with($digit, '8')) {
            $digit = '62' . $digit;
        } else {
            return null;
        }

        // Salah ketik "+62 0812..." -> buang 0 setelah kode negara
        if (str_starts_with($digit, '620')) {
            $digit = '62' . substr($digit, 3);
        }

        return preg_match('/^62\d{8,13}$/', $digit) ? $digit : null;
    }

    private function susunPesan(
        PengajuanMagang $pengajuan,
        array $instansi,
        string $url,
        CarbonInterface $berlakuSampai
    ): string {
        $batas = $berlakuSampai->copy()
            ->timezone('Asia/Jakarta')
            ->locale('id')
            ->translatedFormat('j F Y, H:i');

        $baris = [
            "Halo {$pengajuan->user->name},",
            '',
            'Surat Pengantar Magang untuk pengajuan Anda telah terbit.',
            '',
            "Instansi: {$instansi['nama']}",
        ];

        if (filled($instansi['nomor'])) {
            $baris[] = "Nomor surat: {$instansi['nomor']}";
        }

        if ($instansi['mulai'] && $instansi['selesai']) {
            $baris[] = 'Periode magang: ' . $this->tanggal($instansi['mulai'])
                . ' s.d. ' . $this->tanggal($instansi['selesai']);
        }

        $baris[] = '';
        $baris[] = "Unduh surat melalui tautan berikut (berlaku sampai {$batas} WIB):";
        $baris[] = $url;
        $baris[] = '';
        $baris[] = 'Terima kasih.';

        return implode("\n", $baris);
    }

    private function tanggal(CarbonInterface $tanggal): string
    {
        return $tanggal->copy()->locale('id')->translatedFormat('j F Y');
    }

    private function tolak(string $kolom, string $pesan): never
    {
        throw ValidationException::withMessages([$kolom => $pesan]);
    }
}

<?php

namespace App\Services;

use App\Enums\StatusPengajuan;
use App\Models\PengajuanMagang;
use App\Models\User;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Storage;
use Illuminate\Validation\ValidationException;
use Throwable;

class VerifikasiPengajuanService
{
    public function __construct(
        private NomorSuratService $nomorSurat,
        private SuratMagangService $suratMagang,
    ) {
    }

    /**
     * Admin menyetujui: satu nomor surat dan satu surat pengantar magang untuk TIAP instansi.
     * Satu instansi menghasilkan 1 surat, dua instansi menghasilkan 2 surat bernomor berbeda.
     */
    public function setujui(PengajuanMagang $pengajuan, User $admin): PengajuanMagang
    {
        $pathSurat = [];

        try {
            return DB::transaction(function () use ($pengajuan, $admin, &$pathSurat) {
                $p = $this->kunciDanPastikanMenunggu($pengajuan);

                // Nomor surat: instansi 1 lebih dulu, lalu instansi 2 (nomor berurutan).
                $p->nomor_surat = $this->nomorSurat->berikutnya();
                $p->nomor_surat_kedua = $p->jumlahInstansi() === 2
                    ? $this->nomorSurat->berikutnya()
                    : null;

                $p->status = StatusPengajuan::Setuju;
                $p->diverifikasi_oleh = $admin->id;
                $p->diverifikasi_at = now();

                foreach ($p->daftarInstansi() as $instansi) {
                    $pathSurat[$instansi['urutan']] = $this->suratMagang->simpan($p, $instansi['urutan']);
                }

                $p->file_surat_magang = $pathSurat[1];
                $p->file_surat_magang_kedua = $pathSurat[2] ?? null;
                $p->save();

                return $p;
            });
        } catch (Throwable $e) {
            foreach ($pathSurat as $path) {
                Storage::disk('local')->delete($path);
            }

            throw $e;
        }
    }

    /** Admin meminta revisi: catatan tersimpan, mahasiswa bisa memperbaiki. */
    public function mintaRevisi(PengajuanMagang $pengajuan, User $admin, string $catatan): PengajuanMagang
    {
        return DB::transaction(function () use ($pengajuan, $admin, $catatan) {
            $p = $this->kunciDanPastikanMenunggu($pengajuan);

            $p->status = StatusPengajuan::Revisi;
            $p->catatan_revisi = $catatan;
            $p->diverifikasi_oleh = $admin->id;
            $p->diverifikasi_at = now();
            $p->save();

            return $p;
        });
    }

    /** Ambil ulang barisnya dengan kunci, supaya dua admin tidak memproses pengajuan yang sama. */
    private function kunciDanPastikanMenunggu(PengajuanMagang $pengajuan): PengajuanMagang
    {
        $p = PengajuanMagang::whereKey($pengajuan->id)->lockForUpdate()->firstOrFail();

        if ($p->status !== StatusPengajuan::MenungguVerifikasi) {
            throw ValidationException::withMessages([
                'status' => 'Hanya pengajuan berstatus menunggu verifikasi yang bisa diproses.',
            ]);
        }

        return $p;
    }
}

<?php

namespace App\Support;

class Departemen
{
    /**
     * Prefix NIM (4 karakter pertama) => KODE departemen.
     * Yang disimpan di users.departemen adalah kodenya; nama lengkap dibaca dari
     * config/departemen.php saat ditampilkan (surat, API, dashboard).
     */
    private const PREFIX_NIM = [
        'E141' => 'MNH',
        'E241' => 'HH',
        'E341' => 'KSHE',
        'E441' => 'SVK',
    ];

    /** Nama lengkap dari kode departemen. Kalau kode tidak dikenal, dikembalikan apa adanya. */
    public static function nama(?string $kode): string
    {
        if (! $kode) {
            return '-';
        }

        return config("departemen.{$kode}", $kode);
    }

    /**
     * Nama lengkap untuk respons API. Akun tanpa departemen (misalnya admin)
     * menghasilkan null, bukan "-".
     */
    public static function namaDariKode(?string $kode): ?string
    {
        if (! $kode) {
            return null;
        }

        return self::nama($kode);
    }

    /**
     * Departemen dari prefix NIM (4 karakter pertama, contoh: "E441").
     * Hasil: ['kode' => 'SVK', 'nama' => 'Silvikultur'], atau null kalau prefix tidak dikenal.
     */
    public static function dariNim(?string $nim): ?array
    {
        $prefix = substr(strtoupper(trim((string) $nim)), 0, 4);
        $kode = self::PREFIX_NIM[$prefix] ?? null;

        if (! $kode) {
            return null;
        }

        return [
            'kode' => $kode,
            'nama' => self::nama($kode),
        ];
    }
}
